Final Performance Task (PeTa) - CC102: Travel Expense Tracker

import java.util.Scanner;
public class TravelExpenseTracker {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        double[] expenses = new double[10];
        double total = 0;
        double average;

        System.out.println("=== TRAVEL EXPENSE TRACKER ===");

        for (int i = 0; i < 10; i++) {

            while (true) {
                System.out.print("Enter expense for Trip " + (i + 1) + ": ");

                if (input.hasNextDouble()) {
                    double amount = input.nextDouble();

                    if (amount >= 0) {
                        expenses[i] = amount;
                        total += amount;
                        break;
                    } else {
                        System.out.println("Invalid! Enter nonnegative value.");
                    }

                } else {
                    System.out.println("Invalid input! Enter numbers only.");
                    input.next();
                }
            }
        }

        double highest = expenses[0];
        double lowest = expenses[0];

        int highestTrip = 0;
        int lowestTrip = 0;

        for (int i = 1; i < 10; i++) {

            if (expenses[i] > highest) {
                highest = expenses[i];
                highestTrip = i;
            }

            if (expenses[i] < lowest) {
                lowest = expenses[i];
                lowestTrip = i;
            }
        }
   
        average = total / 10;

        for (int i = 0; i < 9; i++) {
            for (int j = i + 1; j < 10; j++) {

                if (expenses[i] < expenses[j]) {
                    double temp = expenses[i];
                    expenses[i] = expenses[j];
                    expenses[j] = temp;
                }
            }
        }

        System.out.println("\n=== EXPENSE SUMMARY ===");

        System.out.printf("Total Expenses: PHP %.2f%n", total);
        System.out.printf("Average Expense: PHP %.2f%n", average);

        System.out.println("\nMost Expensive Trip:");
        System.out.println("Trip " + (highestTrip + 1));
        System.out.printf("PHP %.2f%n", highest);

        System.out.println("\nLeast Expensive Trip:");
        System.out.println("Trip " + (lowestTrip + 1));
        System.out.printf("PHP %.2f%n", lowest);

        System.out.println("\n=== EXPENSES (DESCENDING ORDER) ===");

        for (int i = 0; i < 10; i++) {
            System.out.printf("Index %d: PHP %.2f%n",
                    i, expenses[i]);
        }
        
        input.close();
    }
}
